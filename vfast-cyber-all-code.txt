// ============================================================================
// HỆ THỐNG TỰ ĐỘNG KHÓA THEO GIỜ QUY ĐỊNH
// Yêu cầu: Khóa vào 21h30 các ngày Thứ 2, 3, 5, 6, 7; các ngày còn lại (Thứ 4, CN) là 22h00
// Mở cửa trở lại lúc 06:00 sáng
// ============================================================================
export interface LockStatusResult {
  isLocked: boolean;
  isEarlyLockDay: boolean; // Thứ 2, 3, 5, 6, 7 (21:30)
  currentDayName: string;
  currentDayIndex: number;
  lockTimeString: string;
  unlockTimeString: string;
  vnTimeString: string;
  vnDateString: string;
  hours: number;
  minutes: number;
  seconds: number;
  remainingSecondsToUnlock?: number;
}

export function calculateLockStatus(date = new Date()): LockStatusResult {
  try {
    // Chuyển đổi sang múi giờ Việt Nam ICT (Asia/Ho_Chi_Minh - UTC+7)
    const vnTimeStr = date.toLocaleString("en-US", { timeZone: "Asia/Ho_Chi_Minh" });
    const vnDate = new Date(vnTimeStr);

    const day = vnDate.getDay(); // 0 = Chủ nhật, 1 = Thứ 2, 2 = Thứ 3, 3 = Thứ 4, 4 = Thứ 5, 5 = Thứ 6, 6 = Thứ 7
    const hours = vnDate.getHours();
    const minutes = vnDate.getMinutes();
    const seconds = vnDate.getSeconds();
    const currentTotalMinutes = hours * 60 + minutes;

    // Ngày khóa 21h30: Thứ 2 (1), Thứ 3 (2), Thứ 5 (4), Thứ 6 (5), Thứ 7 (6)
    // Ngày khóa 22h00: Thứ 4 (3), Chủ Nhật (0)
    const isEarlyLockDay = [1, 2, 4, 5, 6].includes(day);
    const lockMinutes = isEarlyLockDay ? 21 * 60 + 30 : 22 * 60; // 21:30 = 1290p, 22:00 = 1320p
    const unlockMinutes = 6 * 60; // 06:00 sáng = 360p

    // Đang trong khung giờ khóa: Sau giờ đóng cửa (>= lockMinutes) HOẶC trước giờ mở cửa sáng (< unlockMinutes)
    const isLocked = currentTotalMinutes >= lockMinutes || currentTotalMinutes < unlockMinutes;

    const dayNames = ["Chủ Nhật", "Thứ Hai", "Thứ Ba", "Thứ Tư", "Thứ Năm", "Thứ Sáu", "Thứ Bảy"];

    return {
      isLocked,
      isEarlyLockDay,
      currentDayName: dayNames[day],
      currentDayIndex: day,
      lockTimeString: isEarlyLockDay ? "21:30" : "22:00",
      unlockTimeString: "06:00",
      vnTimeString: `${String(hours).padStart(2, "0")}:${String(minutes).padStart(2, "0")}:${String(seconds).padStart(2, "0")}`,
      vnDateString: `${String(vnDate.getDate()).padStart(2, "0")}/${String(vnDate.getMonth() + 1).padStart(2, "0")}/${vnDate.getFullYear()}`,
      hours,
      minutes,
      seconds,
    };
  } catch (e) {
    // Fallback chuẩn nếu môi trường không hỗ trợ timeZone
    const day = date.getDay();
    const hours = date.getHours();
    const minutes = date.getMinutes();
    const seconds = date.getSeconds();
    const currentTotalMinutes = hours * 60 + minutes;
    const isEarlyLockDay = [1, 2, 4, 5, 6].includes(day);
    const lockMinutes = isEarlyLockDay ? 21 * 60 + 30 : 22 * 60;
    const unlockMinutes = 6 * 60;
    const isLocked = currentTotalMinutes >= lockMinutes || currentTotalMinutes < unlockMinutes;

                  className="w-4 h-4 rounded border-[#2b3242] bg-[#181c26] accent-[#00FF66]"
                />
                <span>Tôi đã đọc, hiểu rõ và đồng ý với các điều khoản trên</span>
              </label>

              <button
                onClick={() => {
                  if (tempTermsChecked) {
                    setHasAgreedTerms(true);
                    showToast("Đã xác nhận điều khoản", "Chào mừng bạn đến với VFast Cyber.", "success");
                  }
                }}
                disabled={!tempTermsChecked}
                className={`px-5 py-2 rounded-lg text-xs font-semibold tracking-wide transition-colors cursor-pointer ${
                  tempTermsChecked
                    ? "bg-[#00FF66] text-black hover:bg-[#00e65c]"
                    : "bg-[#181c26] text-[#64748b] cursor-not-allowed border border-[#222734]"
                }`}
              >
                Tôi đã đọc và đồng ý
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ======================================================================
          MODAL BIÊN LAI RÚT TIỀN THÀNH CÔNG (SUCCESS RECEIPT MODAL)
      ====================================================================== */}
      {successReceipt && (
        <div className="fixed inset-0 z-50 bg-black/85 backdrop-blur-sm flex items-center justify-center p-4">
          <div className="w-full max-w-md bg-[#11141c] border border-[#2b3242] rounded-xl shadow-2xl p-6 space-y-5">
            <div className="text-center space-y-1">
              <div className="w-10 h-10 rounded-full bg-[#181c26] border border-[#00FF66]/30 text-[#00FF66] mx-auto flex items-center justify-center mb-2">
                <Icons.CheckCircle />
              </div>
              <h3 className="font-cyber font-bold text-lg text-white">RÚT TIỀN THÀNH CÔNG</h3>
              <p className="text-xs text-[#94a3b8]">Giao dịch đã được hệ thống xác nhận</p>
            </div>

            <div className="p-4 rounded-lg bg-[#181c26] border border-[#222734] space-y-2.5 text-xs">
              <div className="flex justify-between">
                <span className="text-[#64748b]">Mã giao dịch:</span>
                <span className="font-mono font-medium text-white">{successReceipt.id}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Thời gian:</span>
                <span className="text-white">{successReceipt.timestamp}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Ngân hàng nhận:</span>
                <span className="text-white font-medium">{successReceipt.bankName}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Số tiền rút:</span>
                <span className="text-white font-medium">{formatVND(successReceipt.amount)}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Phí xử lý:</span>
                <span className="text-white font-medium">{formatVND(successReceipt.fee)}</span>
              </div>
              <div className="flex justify-between pt-1.5 border-t border-[#222734]">
                <span className="font-semibold text-white">Thực nhận:</span>
                <span className="font-cyber font-bold text-sm text-[#00FF66]">
                  {formatVND(successReceipt.netAmount)}
                </span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Phương thức:</span>
                <span className="text-white">{successReceipt.methodName}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Người nhận:</span>
                <span className="text-white">{successReceipt.recipientName}</span>
              </div>
              <div className="flex justify-between">
                <span className="text-[#64748b]">Số tài khoản/SĐT:</span>
                <span className="text-white">{successReceipt.accountNumber}</span>
              </div>
            </div>

            <div className="flex items-center gap-3">
              <button
                onClick={() => handleDownloadBill(successReceipt)}
                className="flex-1 py-2.5 rounded-lg bg-[#00FF66] text-black font-semibold text-xs flex items-center justify-center gap-2 hover:bg-[#00e65c] transition-colors cursor-pointer"
              >
                <Icons.Download />
                <span>Lưu ảnh bill</span>
              </button>
              <button
                onClick={() => setSuccessReceipt(null)}
                className="px-4 py-2.5 rounded-lg bg-[#181c26] text-[#94a3b8] hover:text-white border border-[#2b3242] text-xs font-medium cursor-pointer"
              >
                Đóng
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ======================================================================
          MODAL XEM & SAO CHÉP TOÀN BỘ CODE TRỰC TIẾP TRÊN ĐIỆN THOẠI
      ====================================================================== */}
      {showAllCodeModal && (
        <div className="fixed inset-0 z-50 bg-black/90 backdrop-blur-md flex items-center justify-center p-3 sm:p-5">
          <div className="w-full max-w-4xl bg-[#11141c] border border-[#2b3242] rounded-xl shadow-2xl flex flex-col max-h-[92vh]">
            <div className="p-4 sm:p-5 border-b border-[#222734] flex items-center justify-between gap-3">
              <div className="flex items-center gap-2.5 min-w-0">
                <div className="w-8 h-8 rounded-lg bg-[#181c26] border border-[#00FF66]/30 text-[#00FF66] flex items-center justify-center shrink-0">
                  <Icons.Code />
                </div>
                <div className="min-w-0">
                  <h3 className="font-cyber font-bold text-sm sm:text-base text-white truncate">
                    TOÀN BỘ MÃ NGUỒN GỘP (1 FILE DUY NHẤT)
                  </h3>
                  <p className="text-[11px] text-[#94a3b8] truncate">
                    Tệp: vfast-cyber-all-code.txt • Đầy đủ data.ts, App.tsx, CSS, HTML, package.json
                  </p>
                </div>
              </div>

              <div className="flex items-center gap-2 shrink-0">
                <button
                  onClick={() => {
                    navigator.clipboard.writeText(allCodeText);
                    showToast("Đã sao chép!", "Toàn bộ mã nguồn đã được lưu vào bộ nhớ tạm của bạn.", "success");
                  }}
                  className="px-3 py-1.5 rounded-lg bg-[#00FF66] text-black font-semibold text-xs hover:bg-[#00e65c] transition-colors flex items-center gap-1.5 cursor-pointer"
                >
                  <Icons.Copy />
                  <span className="hidden sm:inline">Sao chép tất cả</span>
                  <span className="sm:hidden">Chép</span>
                </button>
                <button
                  onClick={() => setShowAllCodeModal(false)}
                  className="p-1.5 rounded-lg bg-[#181c26] text-[#94a3b8] hover:text-white border border-[#2b3242] cursor-pointer"
                >
                  <Icons.Close />
                </button>
              </div>
            </div>

            <div className="p-4 overflow-y-auto flex-1 bg-[#090b10]">
              <div className="mb-3 flex flex-wrap items-center justify-between gap-2 text-xs">
                <div className="text-[#94a3b8] flex items-center gap-2">
                  <span className="w-2 h-2 rounded-full bg-[#00FF66] animate-pulse"></span>
                  <span>Tổng hợp {allCodeText ? (allCodeText.length).toLocaleString() : "135,000+"} ký tự mã nguồn</span>
                </div>
                <a
                  href="/vfast-cyber-all-code.txt"
                  download="vfast-cyber-all-code.txt"
                  onClick={handleDownloadAllCode}
                  className="text-[#00FF66] hover:underline flex items-center gap-1 font-medium cursor-pointer"
                >
                  <Icons.Download />
                  <span>Tải file .txt này về điện thoại</span>
                </a>
              </div>

              <pre className="p-3.5 bg-[#0e1117] text-[#86efac] font-mono text-[11px] leading-relaxed overflow-x-auto rounded-lg border border-[#1f2430] whitespace-pre-wrap select-all selection:bg-[#00FF66] selection:text-black">
                {allCodeText || "Đang tải mã nguồn..."}
              </pre>
            </div>

            <div className="p-3 sm:p-4 border-t border-[#222734] bg-[#0e1117] flex items-center justify-between gap-3">
              <span className="text-[11px] text-[#64748b]">
                Nhấn giữ vào khung code để chọn hoặc bấm "Sao chép tất cả"
              </span>
              <button
                onClick={() => setShowAllCodeModal(false)}
                className="px-4 py-2 rounded-lg bg-[#181c26] text-white border border-[#2b3242] text-xs font-medium hover:bg-[#222734] cursor-pointer"
              >
                Đóng
              </button>
            </div>
          </div>
        </div>
      )}

      {/* ======================================================================
          TOAST NOTIFICATION (GÓC MÀN HÌNH)
      ====================================================================== */}
      <div className="fixed bottom-4 right-4 z-50 space-y-2 max-w-sm w-full pointer-events-none">
        {toasts.map((t) => (
          <div
            key={t.id}
            className="p-3.5 rounded-lg bg-[#181c26] border border-[#2b3242] text-white shadow-lg pointer-events-auto flex items-start gap-2.5 text-xs animate-fadeIn"
          >
            {t.type === "success" && <Icons.CheckCircle />}
            {t.type === "warning" && <Icons.AlertCircle />}
            <div className="min-w-0">
              <div className="font-semibold text-white">{t.title}</div>
              <div className="text-[#94a3b8] mt-0.5">{t.message}</div>
            </div>
          </div>
        ))}
      </div>
    </div>
  );
}
